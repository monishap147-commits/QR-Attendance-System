import { createFileRoute } from "@tanstack/react-router";
import { useEffect, useMemo, useRef, useState } from "react";
import { Camera, Check, Search, Trash2, Pencil, Eye, UserPlus, Power, CameraOff, ScanFace, ShieldCheck } from "lucide-react";
import { toast } from "sonner";
import { Button } from "@/components/ui/button";
import { Input } from "@/components/ui/input";
import { Label } from "@/components/ui/label";
import { Card, CardContent, CardHeader, CardTitle, CardDescription } from "@/components/ui/card";
import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from "@/components/ui/select";
import {
  Table, TableBody, TableCell, TableHead, TableHeader, TableRow,
} from "@/components/ui/table";
import { Badge } from "@/components/ui/badge";
import {
  Dialog, DialogContent, DialogHeader, DialogTitle, DialogDescription,
} from "@/components/ui/dialog";
import {
  AlertDialog, AlertDialogAction, AlertDialogCancel, AlertDialogContent,
  AlertDialogDescription, AlertDialogFooter, AlertDialogHeader, AlertDialogTitle,
} from "@/components/ui/alert-dialog";
import { TagSelect } from "@/components/TagSelect";
import {
  useTaxonomy, addDepartment,
  useStudents, addStudent, removeStudent, toggleStudentStatus,
  getEnrolledFaceEmbeddings,
  type Student,
} from "@/lib/store";
import {
  ensureFaceModelsLoaded, areFaceModelsReady, runLiveFaceEnrollment,
  findDuplicateFace, type EnrollmentStatusUpdate,
} from "@/lib/faceEnrollment";

export const Route = createFileRoute("/_app/students")({
  head: () => ({ meta: [{ title: "Student Management — QRAttend" }] }),
  component: StudentsPage,
});

const emptyForm = {
  id: "", name: "", email: "", phone: "",
  department: "", year: "", section: "",
  photo: "",
};

function StudentsPage() {
  const tax = useTaxonomy();
  const students = useStudents();

  const [form, setForm] = useState({ ...emptyForm });
  const [editingId, setEditingId] = useState<string | null>(null);
  const [cameraOn, setCameraOn] = useState(false);
  const videoRef = useRef<HTMLVideoElement>(null);
  const streamRef = useRef<MediaStream | null>(null);

  const [faceEmbedding, setFaceEmbedding] = useState<number[] | undefined>(undefined);
  const [modelsReady, setModelsReady] = useState(false);
  const [modelsLoading, setModelsLoading] = useState(true);
  const [scanning, setScanning] = useState(false);
  const [scanMessage, setScanMessage] = useState("");
  const [scanProgress, setScanProgress] = useState(0);

  useEffect(() => {
    let cancelled = false;
    ensureFaceModelsLoaded()
      .then(() => {
        if (!cancelled) {
          setModelsReady(true);
          setModelsLoading(false);
        }
      })
      .catch(() => {
        if (!cancelled) {
          setModelsLoading(false);
          toast.error("Failed to load face recognition models. Check your connection and reload.");
        }
      });
    return () => {
      cancelled = true;
    };
  }, []);

  const [search, setSearch] = useState("");
  const [filterDept, setFilterDept] = useState("all");
  const [viewing, setViewing] = useState<Student | null>(null);
  const [deleting, setDeleting] = useState<Student | null>(null);

  const set = <K extends keyof typeof form>(k: K, v: (typeof form)[K]) =>
    setForm((p) => ({ ...p, [k]: v }));

  const startCamera = async () => {
    try {
      const stream = await navigator.mediaDevices.getUserMedia({ video: true, audio: false });
      streamRef.current = stream;
      if (videoRef.current) {
        videoRef.current.srcObject = stream;
        await videoRef.current.play();
      }
      setCameraOn(true);
    } catch {
      toast.error("Unable to access camera");
    }
  };

  const stopCamera = () => {
    streamRef.current?.getTracks().forEach((t) => t.stop());
    streamRef.current = null;
    setCameraOn(false);
  };

  const capture = async () => {
    if (!videoRef.current || !cameraOn) {
      toast.error("Turn the camera on first");
      return;
    }
    if (!modelsReady) {
      toast.error("Face recognition models are still loading. Please wait a moment.");
      return;
    }
    if (scanning) return;

    setScanning(true);
    setScanProgress(0);
    setScanMessage("Starting face scan…");
    try {
      const result = await runLiveFaceEnrollment(videoRef.current, (u: EnrollmentStatusUpdate) => {
        setScanMessage(u.message);
        setScanProgress(u.progress);
      });

      if (!result.success || !result.descriptor || !result.imageDataUrl) {
        toast.error(result.message);
        return;
      }

      const dup = findDuplicateFace(result.descriptor, getEnrolledFaceEmbeddings(editingId ?? undefined));
      if (dup) {
        toast.error("This face is already registered.");
        return;
      }

      set("photo", result.imageDataUrl);
      setFaceEmbedding(result.descriptor);
      toast.success("Face captured and verified");
    } catch {
      toast.error("Face scan failed. Please try again.");
    } finally {
      setScanning(false);
      setScanMessage("");
      setScanProgress(0);
    }
  };

  const resetForm = () => {
    setForm({ ...emptyForm });
    setEditingId(null);
    setFaceEmbedding(undefined);
  };

  const onSubmit = (e: React.FormEvent) => {
    e.preventDefault();
    if (!form.id || !form.name || !form.department || !form.year || !form.section) {
      toast.error("Please fill in all required fields");
      return;
    }
    if (!faceEmbedding || faceEmbedding.length === 0) {
      toast.error("Please complete the face scan before registering.");
      return;
    }
    const existing = students.find((s) => s.id === form.id);
    if (existing && existing.id !== editingId) {
      toast.error("Student ID already exists");
      return;
    }
    const dup = findDuplicateFace(faceEmbedding, getEnrolledFaceEmbeddings(editingId ?? undefined));
    if (dup) {
      toast.error("This face is already registered.");
      return;
    }
    addStudent({
      id: form.id,
      name: form.name,
      email: form.email,
      phone: form.phone,
      department: form.department,
      year: form.year,
      section: form.section,
      subject: "",
      photo: form.photo,
      faceEmbedding,
      registeredAt: existing?.registeredAt ?? new Date().toISOString(),
      status: existing?.status ?? "Active",
    });
    toast.success(editingId ? "Student updated" : `Registered ${form.name}`);
    resetForm();
  };

  const startEdit = (s: Student) => {
    setEditingId(s.id);
    setForm({
      id: s.id, name: s.name, email: s.email ?? "", phone: s.phone ?? "",
      department: s.department, year: s.year, section: s.section,
      photo: s.photo ?? "",
    });
    setFaceEmbedding(s.faceEmbedding);
    window.scrollTo({ top: 0, behavior: "smooth" });
  };

  const filtered = useMemo(() => {
    const q = search.trim().toLowerCase();
    return students.filter((s) => {
      if (filterDept !== "all" && s.department !== filterDept) return false;
      if (!q) return true;
      return `${s.id} ${s.name} ${s.email ?? ""}`.toLowerCase().includes(q);
    });
  }, [students, search, filterDept]);

  return (
    <div className="max-w-7xl mx-auto space-y-6">
      <div>
        <h2 className="text-2xl font-bold tracking-tight">
          {editingId ? "Edit student" : "Register a new student"}
        </h2>
        <p className="text-sm text-muted-foreground">
          Fill in the details and capture the student's face for enrollment.
        </p>
      </div>

      <form className="grid gap-6 lg:grid-cols-3" onSubmit={onSubmit}>
        <div className="lg:col-span-2 space-y-6">
          <Card className="glass-card">
            <CardHeader>
              <CardTitle className="text-base">Personal Information</CardTitle>
              <CardDescription>Basic contact details.</CardDescription>
            </CardHeader>
            <CardContent className="grid gap-4 sm:grid-cols-2">
              <Field label="Student ID *" id="sid">
                <Input id="sid" value={form.id} onChange={(e) => set("id", e.target.value)}
                  disabled={!!editingId} placeholder="e.g. CS2024001" />
              </Field>
              <Field label="Full Name *" id="name">
                <Input id="name" value={form.name} onChange={(e) => set("name", e.target.value)} placeholder="Jane Doe" />
              </Field>
              <Field label="Email" id="email">
                <Input id="email" type="email" value={form.email}
                  onChange={(e) => set("email", e.target.value)} placeholder="student@college.edu" />
              </Field>
              <Field label="Phone Number" id="phone">
                <Input id="phone" type="tel" value={form.phone}
                  onChange={(e) => set("phone", e.target.value)} placeholder="+1 555 0100" />
              </Field>
            </CardContent>
          </Card>

          <Card className="glass-card">
            <CardHeader>
              <CardTitle className="text-base">Academic Details</CardTitle>
              <CardDescription>Type to add a new department.</CardDescription>
            </CardHeader>
            <CardContent className="grid gap-4 sm:grid-cols-2 lg:grid-cols-3">
              <Field label="Department *">
                <TagSelect
                  value={form.department}
                  onChange={(v) => setForm((p) => ({ ...p, department: v }))}
                  options={tax.departments}
                  onCreate={addDepartment}
                  placeholder="Select or type…"
                />
              </Field>
              <Field label="Year *">
                <Select value={form.year} onValueChange={(v) => setForm((p) => ({ ...p, year: v }))}>
                  <SelectTrigger><SelectValue placeholder="Select" /></SelectTrigger>
                  <SelectContent>
                    {[1, 2, 3, 4].map((y) => (
                      <SelectItem key={y} value={String(y)}>Year {y}</SelectItem>
                    ))}
                  </SelectContent>
                </Select>
              </Field>
              <Field label="Section *">
                <Select value={form.section} onValueChange={(v) => set("section", v)}>
                  <SelectTrigger><SelectValue placeholder="Select" /></SelectTrigger>
                  <SelectContent>
                    {["A", "B", "C", "D"].map((s) => (
                      <SelectItem key={s} value={s}>Section {s}</SelectItem>
                    ))}
                  </SelectContent>
                </Select>
              </Field>
            </CardContent>
          </Card>
        </div>

        <div className="space-y-4">
          <Card className="glass-card">
            <CardHeader>
              <CardTitle className="text-base">Face Capture</CardTitle>
              <CardDescription>
                {modelsLoading
                  ? "Loading face recognition models…"
                  : "Turn on the camera, then scan to enroll a live face."}
              </CardDescription>
            </CardHeader>
            <CardContent className="space-y-3">
              <div className="relative aspect-square rounded-xl border-2 border-dashed bg-muted/40 grid place-items-center overflow-hidden">
                <video ref={videoRef} playsInline muted
                  className={`absolute inset-0 h-full w-full object-cover ${cameraOn ? "block" : "hidden"}`} />
                {!cameraOn && !form.photo && (
                  <div className="text-center text-muted-foreground">
                    <Camera className="h-10 w-10 mx-auto mb-2 opacity-60" />
                    <p className="text-xs">Camera off</p>
                  </div>
                )}
                {!cameraOn && form.photo && (
                  // eslint-disable-next-line jsx-a11y/alt-text
                  <img src={form.photo} alt="" className="absolute inset-0 h-full w-full object-cover" />
                )}
                {cameraOn && !scanning && (
                  <div className="absolute inset-6 rounded-full border-2 border-primary/40 pointer-events-none" />
                )}
                {cameraOn && scanning && (
                  <div className="absolute inset-0 flex flex-col items-center justify-end bg-black/30 p-3 gap-2">
                    <p className="text-xs font-medium text-white text-center drop-shadow">{scanMessage}</p>
                    <div className="h-1.5 w-full rounded-full bg-white/30 overflow-hidden">
                      <div
                        className="h-full bg-primary transition-all"
                        style={{ width: `${Math.round(scanProgress * 100)}%` }}
                      />
                    </div>
                  </div>
                )}
                {!scanning && faceEmbedding && form.photo && (
                  <div className="absolute top-2 right-2 rounded-full bg-success text-success-foreground p-1" title="Face verified">
                    <ShieldCheck className="h-4 w-4" />
                  </div>
                )}
              </div>
              <div className="grid grid-cols-2 gap-2">
                {cameraOn ? (
                  <Button type="button" variant="secondary" onClick={stopCamera} disabled={scanning} className="gap-2">
                    <CameraOff className="h-4 w-4" /> Camera Off
                  </Button>
                ) : (
                  <Button type="button" variant="secondary" onClick={startCamera} className="gap-2">
                    <Camera className="h-4 w-4" /> Camera On
                  </Button>
                )}
                <Button type="button" onClick={capture} disabled={!cameraOn || scanning || !modelsReady} className="gap-2">
                  <ScanFace className="h-4 w-4" /> {scanning ? "Scanning…" : "Scan Face"}
                </Button>
              </div>
              <p className="text-xs text-muted-foreground">
                Look at the camera and blink naturally. We verify liveness and reject photos, screens, and duplicate faces.
              </p>
            </CardContent>
          </Card>

          <div className="flex flex-col gap-2">
            <Button type="submit" className="w-full gap-2" size="lg" disabled={scanning}>
              <Check className="h-4 w-4" /> {editingId ? "Save changes" : "Register Student"}
            </Button>
            {editingId && (
              <Button type="button" variant="ghost" onClick={resetForm}>Cancel edit</Button>
            )}
          </div>
        </div>
      </form>

      <Card className="glass-card">
        <CardHeader className="flex flex-row flex-wrap items-end justify-between gap-3">
          <div>
            <CardTitle className="text-base flex items-center gap-2">
              <UserPlus className="h-4 w-4 text-primary" /> Student List
            </CardTitle>
            <CardDescription>{filtered.length} of {students.length} students</CardDescription>
          </div>
          <div className="flex flex-wrap gap-2 items-end">
            <div className="relative">
              <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
              <Input value={search} onChange={(e) => setSearch(e.target.value)}
                placeholder="Search by ID, name or email…" className="pl-9 w-64" />
            </div>
            <Select value={filterDept} onValueChange={setFilterDept}>
              <SelectTrigger className="w-44"><SelectValue placeholder="Department" /></SelectTrigger>
              <SelectContent>
                <SelectItem value="all">All Departments</SelectItem>
                {tax.departments.map((d) => (
                  <SelectItem key={d} value={d}>{d}</SelectItem>
                ))}
              </SelectContent>
            </Select>
          </div>
        </CardHeader>
        <CardContent className="p-0">
          <div className="overflow-x-auto">
            <Table>
              <TableHeader>
                <TableRow>
                  <TableHead>Student ID</TableHead>
                  <TableHead>Name</TableHead>
                  <TableHead>Department</TableHead>
                  <TableHead>Year</TableHead>
                  <TableHead>Section</TableHead>
                  <TableHead>Status</TableHead>
                  <TableHead className="text-right">Actions</TableHead>
                </TableRow>
              </TableHeader>
              <TableBody>
                {filtered.length === 0 ? (
                  <TableRow>
                    <TableCell colSpan={7} className="text-center text-sm text-muted-foreground py-12">
                      No students yet. Register one above to see them here.
                    </TableCell>
                  </TableRow>
                ) : (
                  filtered.map((s) => (
                    <TableRow key={s.id}>
                      <TableCell className="font-mono text-xs">{s.id}</TableCell>
                      <TableCell className="font-medium">{s.name}</TableCell>
                      <TableCell>{s.department}</TableCell>
                      <TableCell>Year {s.year}</TableCell>
                      <TableCell>{s.section}</TableCell>
                      <TableCell>
                        <Badge variant={s.status === "Inactive" ? "secondary" : "default"}
                          className={s.status === "Inactive" ? "" : "bg-success text-success-foreground hover:bg-success/90"}>
                          {s.status ?? "Active"}
                        </Badge>
                      </TableCell>
                      <TableCell className="text-right">
                        <div className="flex justify-end gap-1">
                          <Button variant="ghost" size="icon" className="h-8 w-8" onClick={() => setViewing(s)} title="View">
                            <Eye className="h-4 w-4" />
                          </Button>
                          <Button variant="ghost" size="icon" className="h-8 w-8" onClick={() => startEdit(s)} title="Edit">
                            <Pencil className="h-4 w-4" />
                          </Button>
                          <Button variant="ghost" size="icon" className="h-8 w-8" onClick={() => toggleStudentStatus(s.id)} title="Toggle status">
                            <Power className="h-4 w-4" />
                          </Button>
                          <Button variant="ghost" size="icon" className="h-8 w-8 text-destructive hover:text-destructive" onClick={() => setDeleting(s)} title="Delete">
                            <Trash2 className="h-4 w-4" />
                          </Button>
                        </div>
                      </TableCell>
                    </TableRow>
                  ))
                )}
              </TableBody>
            </Table>
          </div>
        </CardContent>
      </Card>

      <Dialog open={!!viewing} onOpenChange={(o) => !o && setViewing(null)}>
        <DialogContent>
          <DialogHeader>
            <DialogTitle>{viewing?.name}</DialogTitle>
            <DialogDescription>Student profile</DialogDescription>
          </DialogHeader>
          {viewing && (
            <div className="grid gap-4 sm:grid-cols-[auto_1fr] items-start">
              <div className="h-28 w-28 rounded-xl bg-muted overflow-hidden grid place-items-center">
                {viewing.photo ? (
                  // eslint-disable-next-line jsx-a11y/alt-text
                  <img src={viewing.photo} alt="" className="h-full w-full object-cover" />
                ) : (
                  <Camera className="h-8 w-8 text-muted-foreground" />
                )}
              </div>
              <div className="space-y-1 text-sm">
                <Row k="Student ID" v={viewing.id} />
                <Row k="Email" v={viewing.email || "—"} />
                <Row k="Phone" v={viewing.phone || "—"} />
                <Row k="Department" v={viewing.department} />
                <Row k="Year / Section" v={`Year ${viewing.year} · ${viewing.section}`} />
                <Row k="Status" v={viewing.status ?? "Active"} />
              </div>
            </div>
          )}
        </DialogContent>
      </Dialog>

      <AlertDialog open={!!deleting} onOpenChange={(o) => !o && setDeleting(null)}>
        <AlertDialogContent>
          <AlertDialogHeader>
            <AlertDialogTitle>Delete {deleting?.name}?</AlertDialogTitle>
            <AlertDialogDescription>This action cannot be undone.</AlertDialogDescription>
          </AlertDialogHeader>
          <AlertDialogFooter>
            <AlertDialogCancel>Cancel</AlertDialogCancel>
            <AlertDialogAction
              className="bg-destructive text-destructive-foreground hover:bg-destructive/90"
              onClick={() => {
                if (deleting) {
                  removeStudent(deleting.id);
                  toast.success("Student deleted");
                }
                setDeleting(null);
              }}
            >
              Delete
            </AlertDialogAction>
          </AlertDialogFooter>
        </AlertDialogContent>
      </AlertDialog>
    </div>
  );
}

function Field({ label, id, children }: { label: string; id?: string; children: React.ReactNode }) {
  return (
    <div className="space-y-2">
      <Label htmlFor={id}>{label}</Label>
      {children}
    </div>
  );
}

function Row({ k, v }: { k: string; v: string }) {
  return (
    <div className="flex justify-between gap-3">
      <span className="text-muted-foreground">{k}</span>
      <span className="font-medium text-right truncate">{v}</span>
    </div>
  );
}
